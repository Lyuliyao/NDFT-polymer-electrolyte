  \ifx\AP@\relax\else
    \global\output\expandafter{\the\AP@output}%
    \global\expandafter\let\expandafter\AP@\expandafter\relax\AP@
  \fi
  \clearpage
}{}{\PackageError{pnas-firstpage}{Could not patch \string\Parasplit}{Check the PNAS template version.}}
